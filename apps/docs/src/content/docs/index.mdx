import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Two ways to write" icon="pencil">
    Long posts and glossary terms are Markdown files in a Nextcloud folder
    (Obsidian works well). Short notes are Memos.
  </Card>
  <Card title="Static and fast" icon="rocket">
    A builder turns both sources into plain HTML. No database, no server code at
    runtime, a strict Content Security Policy without foreign sources.
  </Card>
  <Card title="Search, RSS, glossary" icon="magnifier">
    Full-text search that runs in the browser, one RSS feed, and a glossary with
    a letter view.
  </Card>
  <Card title="Comments" icon="comment">
    Self-hosted Remark42 with login by email only. The comments are also part of
    the page, so they read without JavaScript.
  </Card>
  <Card title="One command" icon="seti:docker">
    `docker compose up -d` behind your reverse proxy, configured in one `.env`
    file. A local demo with sample content is one command too.
  </Card>
  <Card title="Safe by default" icon="approve-check">
    Content is sanitized, a failed build never replaces the last good version,
    secrets stay in `.env`.
  </Card>
</CardGrid>
